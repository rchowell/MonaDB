= Select Statement

The Select Statement _maps_ the input bindings and applies the constructor; the constructor may be an object or a list of select items.

.TODOS

.Examples
[source,sql]
----
select 1 + 1;

-- equivalences

select { p.x } from points as p;
select p.x as x from points as p;

select { p.x, p.y } from points as p;
select p.x as x, p.y as y from points as p;
select { x: p.x, y: p.y } from points as p;

select * from t;
select { ...t } from t;

select * from t, s;
select { ...t, ...s } from t, s;
----

== Syntax

[source,mckeeman]
----
select_statement
    "select" select_constructor
    "select" select_constructor select_body

select_constructor
    "."
    "*"
    select_expr
    select_list

select_expr
    expr

select_list
    select_item
    select_item "," select_list

select_item
    expr "as" ident

select_path
    expr_var
    expr_jpk
----

== Proposals

.select/delete one
[source,sql]
----
-- select one
select T@<id>;

-- with constructor
select upper(t.name) from T@<id> as t;

-- delete one
delete T@<id>;
----

== Spread Operator

[source,sql]
----
select * from T     -- map vars.*   / spread
select . from T     -- map vars     / identity
----

.Equivalences
[source]
----
*           <=> { ... }
a.*         <=> { ...a }
a.*, b.*    <=> { ...a, ...b }
----

.Example 1
[source,sql]
----
-- input: T = [ { x: 1, y: 2 } ]

select * from T as t;           -- vars { t: t }
select t.* from T as t;         -- ..
select { ...t } from T as t;    -- vars { x: t.x, y: t.y }

-- output: [ { x: 1, y: 2 } ]
----

.Example 2
[source]
----
-- input: T = [ { x: 1, y: 2 } ]

select . from T as t with *;         -- vars { x: t.x, y: t.y }
select . from T as t with t.*;       -- ..
select . from T as t with { ...t };  -- ..

-- output: [ { x: 1, y: 2 } ]
----

.Example 3
[source,sql]
----
-- input: T = [ { x: 1, y: 2 } ]

select * from T as t;           -- vars { t: t }
select t.* from T as t;         -- ..
select { ...t } from T as t;    -- vars { x: t.x, y: t.y }

-- output: [ { x: 1, y: 2 } ]
----
