= Select Statement

.Example
[source,rql]
----
select p.x, p.y from points as p;

-- equivalences

select   * from v;
select v.* from v;
select { ...v } from v;
----

== Syntax

[source,mckeeman]
----
select_statement
    "select" select_constructor "from" table_name

select_constructor
    "*"
    select_obj
    select_list

select_object
    expr_obj

select_list
    select_item
    select_item "," select_list

select_item
    expr_var
    expr_jpk
    expr "as" ident

select_path
    expr_var
    expr_jpk
----

== Description

The Select Statement _maps_ the input bindings and applies the constructor; the constructor may be an object or a list of select items.
