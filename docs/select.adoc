= Select Statement

.Example
[source,rql]
----
select p.x, p.y from points as p;
----

== Syntax

[source,mckeeman]
----
select_statement
    "SELECT" select_list "FROM" table_name

select_list
    select_item
    select_item "," select_list

select_item
    expr "AS" ident
----
