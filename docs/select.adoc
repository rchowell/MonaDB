= Select Statement

The Select Statement _maps_ the input bindings and applies the constructor; the constructor may be an object or a list of select items.

.Examples
[source,rql]
----
-- basic
select 1 + 1;

-- equivalences

select { p.x } from points as p;
select p.x as x from points as p;

select { p.x, p.y } from points as p;
select p.x as x, p.y as y from points as p;
select { x: p.x, y: p.y } from points as p;

select * from t;
select { ...t } from t;

select * from t, s;
select { ...t, ...s } from t, s;
----

== Syntax

[source,mckeeman]
----
select_statement
    "select" select_constructor "from" table_name

select_constructor
    "*"
    select_obj
    select_list

select_object
    expr_obj

select_list
    select_item
    select_item "," select_list

select_item
    expr "as" ident

select_path
    expr_var
    expr_jpk
----

== Proposals

[source,rql]
----
-- select one
select * from T@<id>;

-- with constructor
select upper(t.name) from T@<id> as t;

-- delete one
delete from T@<id>;
----
